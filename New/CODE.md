```ts
import { NextRequest, NextResponse } from "next/server";
import { saveConfig } from "@/lib/config";
import { makeOAuthClient, whoAmI } from "@/lib/sheets";

export const dynamic = "force-dynamic";

/**
 * GET /api/auth/callback — completes Google sign-in.
 * Exchanges the code for tokens, stores the refresh token locally, and sends
 * the user back to the app (the wizard continues to the sheet-choice step).
 */
export async function GET(req: NextRequest) {
  const url = new URL(req.url);
  const code = url.searchParams.get("code");
  const state = url.searchParams.get("state");
  const cookieState = req.cookies.get("oauth_state")?.value;

  const back = (param: string) => {
    const res = NextResponse.redirect(`${url.origin}/?${param}`);
    res.cookies.delete("oauth_state");
    return res;
  };

  if (url.searchParams.get("error")) return back("auth=denied");
  if (!code || !state || state !== cookieState) return back("auth=invalid");

  try {
    const client = makeOAuthClient(`${url.origin}/api/auth/callback`);
    const { tokens } = await client.getToken(code);
    if (!tokens.refresh_token) return back("auth=noRefresh");

    client.setCredentials(tokens);
    let email = "";
    try {
      email = await whoAmI(client);
    } catch {
      /* email is cosmetic */
    }

    saveConfig({
      authType: "oauth",
      refreshToken: tokens.refresh_token,
      userEmail: email,
      // Clear any stale SA credentials from previous setups.
      serviceAccountEmail: "",
      privateKey: "",
    });
    return back("auth=ok");
  } catch {
    return back("auth=failed");
  }
}
```

```ts
import { NextRequest, NextResponse } from "next/server";
import crypto from "crypto";
import { getConfig } from "@/lib/config";
import { OAUTH_SCOPES, makeOAuthClient } from "@/lib/sheets";

export const dynamic = "force-dynamic";

/**
 * GET /api/auth/login — start the Google sign-in flow.
 * Redirects the browser to Google's consent screen; state is kept in an
 * httpOnly cookie to protect the callback against CSRF.
 */
export async function GET(req: NextRequest) {
  const cfg = getConfig();
  if (!cfg.oauthClientId) {
    return NextResponse.json(
      { error: "חסר oauth-client.json — ראה README, סעיף הפצה" },
      { status: 500 }
    );
  }

  const origin = new URL(req.url).origin;
  const redirectUri = `${origin}/api/auth/callback`;
  const state = crypto.randomUUID();

  const url = makeOAuthClient(redirectUri).generateAuthUrl({
    access_type: "offline",
    prompt: "consent", // guarantees a refresh_token on every connect
    scope: OAUTH_SCOPES,
    state,
  });

  const res = NextResponse.redirect(url);
  res.cookies.set("oauth_state", state, {
    httpOnly: true,
    sameSite: "lax",
    maxAge: 600,
    path: "/",
  });
  return res;
}
```

```ts
import { NextRequest, NextResponse } from "next/server";
import { getMonthData, saveReport } from "@/lib/data";
import { CLASSIFICATIONS, ReportInput } from "@/lib/types";

/** Values that may legally be written to the classification column. */
const ALLOWED_CLASSIFICATIONS = new Set<string>([
  ...CLASSIFICATIONS,
  "סופשבוע",
  "חול המועד",
]);

function toMinutes(t: string): number {
  const [h, m] = t.split(":").map(Number);
  return h * 60 + m;
}

function isValidAbsence(v: unknown): boolean {
  return (
    v === undefined ||
    (typeof v === "number" && Number.isFinite(v) && v >= 0 && v <= 1)
  );
}

export const dynamic = "force-dynamic";

/** GET /api/reports?year=2026&month=7 */
export async function GET(req: NextRequest) {
  const { searchParams } = new URL(req.url);
  const now = new Date();
  const year = parseInt(searchParams.get("year") ?? "", 10) || now.getFullYear();
  const month = parseInt(searchParams.get("month") ?? "", 10) || now.getMonth() + 1;
  const force = searchParams.get("force") === "1";
  try {
    const data = await getMonthData(year, month, force);
    return NextResponse.json(data);
  } catch (e) {
    return NextResponse.json(
      { error: e instanceof Error ? e.message : "שגיאה בקריאת הנתונים" },
      { status: 500 }
    );
  }
}

/** POST /api/reports — upsert a single day report (raw fields only). */
export async function POST(req: NextRequest) {
  let body: ReportInput;
  try {
    body = (await req.json()) as ReportInput;
  } catch {
    return NextResponse.json({ error: "גוף בקשה לא תקין" }, { status: 400 });
  }

  if (!body.date || !/^\d{4}-\d{2}-\d{2}$/.test(body.date)) {
    return NextResponse.json({ error: "תאריך חסר או לא תקין" }, { status: 400 });
  }
  const hhmm = /^([01]?\d|2[0-3]):[0-5]\d$/;
  if (body.entry && !hhmm.test(body.entry)) {
    return NextResponse.json({ error: "שעת כניסה לא תקינה (HH:mm)" }, { status: 400 });
  }
  if (body.exit && !hhmm.test(body.exit)) {
    return NextResponse.json({ error: "שעת יציאה לא תקינה (HH:mm)" }, { status: 400 });
  }
  if ((body.entry && !body.exit) || (!body.entry && body.exit)) {
    return NextResponse.json({ error: "יש למלא גם כניסה וגם יציאה" }, { status: 400 });
  }
  if (body.entry && body.exit && toMinutes(body.exit) <= toMinutes(body.entry)) {
    return NextResponse.json(
      { error: "שעת היציאה חייבת להיות אחרי שעת הכניסה" },
      { status: 400 }
    );
  }
  if (body.classification && !ALLOWED_CLASSIFICATIONS.has(body.classification)) {
    return NextResponse.json({ error: "סיווג לא מוכר" }, { status: 400 });
  }
  if (
    !isValidAbsence(body.vacationDays) ||
    !isValidAbsence(body.sickDays) ||
    !isValidAbsence(body.reserveDays)
  ) {
    return NextResponse.json(
      { error: "ערך היעדרות חייב להיות בין 0 ל-1" },
      { status: 400 }
    );
  }
  if (typeof body.notes === "string" && body.notes.length > 500) {
    return NextResponse.json({ error: "הערה ארוכה מדי (עד 500 תווים)" }, { status: 400 });
  }

  try {
    const record = await saveReport(body);
    return NextResponse.json({ record });
  } catch (e) {
    return NextResponse.json(
      { error: e instanceof Error ? e.message : "שגיאה בשמירת הדיווח" },
      { status: 500 }
    );
  }
}
```

## `app/api/setup/route.ts`

```ts
import { NextRequest, NextResponse } from "next/server";
import { getConfig } from "@/lib/config";
import {
  commitSetup,
  createNewSpreadsheet,
  makeSetupAuth,
  validateCredentials,
  validateExistingSheet,
  whoAmI,
  type SetupCredentials,
} from "@/lib/sheets";

export const dynamic = "force-dynamic";

/** GET /api/setup — onboarding status (never leaks secrets). */
export async function GET() {
  const cfg = getConfig();
  return NextResponse.json({
    configured: cfg.configured,
    sheetUrl: cfg.sheetUrl,
    serviceAccountEmail: cfg.serviceAccountEmail,
    userEmail: cfg.userEmail,
    computeDerived: cfg.computeDerived,
    authType: cfg.authType,
    /** oauth-client.json is bundled -> the Google button can be shown. */
    oauthReady: Boolean(cfg.oauthClientId),
    /** The user finished Google sign-in (tokens stored). */
    googleConnected: Boolean(cfg.refreshToken),
  });
}

interface SetupBody {
  /** Optional — advanced mode. When omitted, the stored OAuth login is used. */
  serviceAccountJson?: string;
  choice: "new" | "existing";
  sheetIdOrUrl?: string;
  userEmail?: string;
  folderUrl?: string;
  title?: string;
}

function extractSheetId(input: string): string {
  const m = /\/spreadsheets\/d\/([a-zA-Z0-9_-]{20,})/.exec(input);
  if (m) return m[1];
  if (/^[a-zA-Z0-9_-]{20,}$/.test(input.trim())) return input.trim();
  return "";
}

function extractFolderId(input: string): string {
  const m = /\/folders\/([a-zA-Z0-9_-]{10,})/.exec(input);
  if (m) return m[1];
  if (/^[a-zA-Z0-9_-]{10,}$/.test(input.trim())) return input.trim();
  return "";
}

/** POST /api/setup — connect existing sheet or create a new one. */
export async function POST(req: NextRequest) {
  let body: SetupBody;
  try {
    body = (await req.json()) as SetupBody;
  } catch {
    return NextResponse.json({ error: "גוף בקשה לא תקין" }, { status: 400 });
  }

  // Resolve identity: uploaded Service Account JSON (advanced) or the
  // signed-in Google user (default distribution flow).
  let creds: SetupCredentials | undefined;
  if (body.serviceAccountJson) {
    try {
      const sa = JSON.parse(body.serviceAccountJson);
      creds = {
        clientEmail: String(sa.client_email ?? ""),
        privateKey: String(sa.private_key ?? ""),
      };
    } catch {
      return NextResponse.json(
        { error: "קובץ ה-Service Account אינו JSON תקין" },
        { status: 400 }
      );
    }
    if (!creds.clientEmail || !creds.privateKey.includes("BEGIN PRIVATE KEY")) {
      return NextResponse.json(
        { error: "ה-JSON חייב להכיל client_email ו-private_key" },
        { status: 400 }
      );
    }
  }

  let auth;
  try {
    auth = makeSetupAuth(creds);
    await validateCredentials(auth);
  } catch (e) {
    const msg = creds
      ? "אימות מול Google נכשל — ודא שה-Sheets וה-Drive API מופעלים בפרויקט"
      : e instanceof Error
        ? e.message
        : "יש להתחבר ל-Google תחילה";
    return NextResponse.json({ error: msg }, { status: 400 });
  }

  const authType: "sa" | "oauth" = creds ? "sa" : "oauth";
  const saPatch = creds
    ? { serviceAccountEmail: creds.clientEmail, privateKey: creds.privateKey }
    : {};

  try {
    if (body.choice === "existing") {
      const sheetId = extractSheetId(body.sheetIdOrUrl ?? "");
      if (!sheetId) {
        return NextResponse.json({ error: "קישור/מזהה גיליון לא תקין" }, { status: 400 });
      }
      const { firstTab } = await validateExistingSheet(auth, sheetId);
      commitSetup({
        sheetId,
        sheetUrl: `https://docs.google.com/spreadsheets/d/${sheetId}/edit`,
        authType,
        ...saPatch,
        userEmail: body.userEmail || getConfig().userEmail,
        computeDerived: false,
        sheetTab: firstTab,
      });
      return NextResponse.json({
        ok: true,
        mode: "existing",
        sheetUrl: `https://docs.google.com/spreadsheets/d/${sheetId}/edit`,
      });
    }

    // New sheet. With OAuth the user owns it (no sharing, no folder needed);
    // with an SA it is created under the SA and shared to the user.
    const folderId = body.folderUrl ? extractFolderId(body.folderUrl) : "";
    if (body.folderUrl && !folderId) {
      return NextResponse.json({ error: "קישור תיקיית Drive לא תקין" }, { status: 400 });
    }
    const created = await createNewSpreadsheet({
      auth,
      title: body.title?.trim() || `דיווח שעות ${new Date().getFullYear()}`,
      shareWith: authType === "sa" ? body.userEmail?.trim() || undefined : undefined,
      folderId: folderId || undefined,
    });
    let userEmail = body.userEmail?.trim() || getConfig().userEmail;
    if (!userEmail) {
      try {
        userEmail = await whoAmI(auth);
      } catch {
        /* cosmetic */
      }
    }
    commitSetup({
      sheetId: created.sheetId,
      sheetUrl: created.sheetUrl,
      authType,
      ...saPatch,
      userEmail,
      computeDerived: true,
    });
    return NextResponse.json({ ok: true, mode: "new", sheetUrl: created.sheetUrl });
  } catch (e) {
    const msg = e instanceof Error ? e.message : "שגיאה בהגדרת הגיליון";
    const hint =
      creds && /403|permission|The caller does not have permission/i.test(msg)
        ? ` — ודא שהגיליון/התיקייה משותפים עם ${creds.clientEmail} בהרשאת Editor`
        : "";
    return NextResponse.json({ error: msg + hint }, { status: 500 });
  }
}
```

```tsx
  const [month, setMonth] = useState(now.getMonth() + 1); // 1-12

      setYear(y);
    },

      const json = await res.json();

        await mutate("/api/setup");
        await mutate("/api/reserve");
      }
    } finally {
      setSyncing(false);
    }
  }, [year, month, mutate]);

  const openNewReport = useCallback(() => {
    setEditRecord(null);
    setModalOpen(true);
  }, []);

    setEditRecord(rec);
    setModalOpen(true);
  }, []);

    async (input: ReportInput) => {

      const key = monthKey(y, m);

                      exit: input.exit ?? "",

                      vacationDays: input.vacationDays ?? 0,
                      reserveDays: input.reserveDays ?? 0,
                      sickDays: input.sickDays ?? 0,

        onTab={setTab}
        year={year}
        month={month}
        onPrevMonth={() => shiftMonth(-1)}
        onNextMonth={() => shiftMonth(1)}
        onNewReport={openNewReport}
        onSync={syncNow}
        syncing={syncing}
        onSelectMonth={(y, m) => {
          setYear(y);
          setMonth(m);
        }}
      />

      <main className="mx-auto max-w-7xl px-4 py-6 sm:px-6">
        {error && (
          <div className="mb-4 rounded-xl border border-red-200 bg-red-50 px-4 py-3 text-sm text-red-700">
            {String(error.message ?? error)}
          </div>
        )}

          <Dashboard
            data={data}
            loading={isLoading}
            year={year}
            month={month}
            sheetUrl={setup?.sheetUrl}
            onOpenSetup={() => {
              setSetupSkipped(false);
              setSetupForced(true);
            }}
          />
        )}

          <ReportsScreen
            data={data}
            year={year}
            month={month}
            onYear={setYear}
            onMonth={setMonth}
            onEdit={openEdit}
          />

        {tab === "settings" && <SettingsScreen />}
```

```tsx
  useEffect(() => {
    if (!open) return;

    };
    const onKey = (e: KeyboardEvent) => {
      if (e.key === "Escape") setOpen(false);
    };
    document.addEventListener("mousedown", onDown);
    document.addEventListener("keydown", onKey);
    return () => {
      document.removeEventListener("mousedown", onDown);
      document.removeEventListener("keydown", onKey);
    };
  }, [open]);

  const toggle = () => {
    setStage("years");
    setPickedYear(year);

  };

  const years: number[] = [];
  for (let y = FIRST_YEAR; y <= new Date().getFullYear() + 1; y++) years.push(y);
```

```ts
import type { DayRecord, MonthSummary } from "./types";
import { round1, todayIso } from "./date";

const DEFAULT_DAILY_HOURS = 9;

/**
 * Pure KPI computation for one month of records.
 * targetHours = standard work days (from settings) * 9h, as displayed in the
 * "הגדרות תקן" screen (e.g. July: 22 days -> 198.0h).
 */
export function computeMonthSummary(
  year: number,
  month: number,
  records: DayRecord[],
  standardWorkDays: number
): MonthSummary {
  const targetHours = round1(standardWorkDays * DEFAULT_DAILY_HOURS);
  const reportedHours = round1(
    records.reduce((s, r) => s + (r.totalHoursDecimal || 0), 0)
  );
  const remainingHours = round1(Math.max(targetHours - reportedHours, 0));

  const today = todayIso();

  // Potential remaining workdays: days from today onward with a positive daily
  // standard (i.e. not weekend/holiday) and no hours reported yet.
  const remainingWorkdays = records.filter(
    (r) =>
      r.date >= today &&
      r.dailyStandard > 0 &&
      (r.totalHoursDecimal || 0) === 0 &&
      !r.vacationDays &&
      !r.sickDays &&
      !r.reserveDays
  ).length;

  const forecastHoursPerDay =
    remainingWorkdays > 0 ? round1(remainingHours / remainingWorkdays) : 0;

  const vacationUsedThisMonth = round1(
    records.reduce((s, r) => s + (r.vacationDays || 0), 0)
  );
  const sickDaysThisMonth = round1(
    records.reduce((s, r) => s + (r.sickDays || 0), 0)
  );
  const reserveDaysThisMonth = round1(
    records.reduce((s, r) => s + (r.reserveDays || 0), 0)
  );

  // Latest non-zero cumulative vacation balance within the month.
  let vacationBalance = 0;
  for (const r of records) {
    if (r.vacationBalance) vacationBalance = r.vacationBalance;
  }
  vacationBalance = round1(vacationBalance);

  return {
    year,
    month,
    targetHours,
    reportedHours,
    remainingHours,
    remainingWorkdays,
    forecastHoursPerDay,
    approxRemainingDays: round1(remainingHours / DEFAULT_DAILY_HOURS),
    vacationUsedThisMonth,
    sickDaysThisMonth,
    reserveDaysThisMonth,
    vacationBalance,
    standardWorkDays,
  };
}
```

```ts
  const live = sheets.isLiveMode();
  const all = live ? await sheets.readJournal(force) : mock.mockAllRecords();

  const perYear = new Map<number, ReserveBucket>();
  const perMonth = new Map<number, Map<number, ReserveBucket>>();

  for (const r of all) {
    if (!r.reserveDays) continue;
    const year = parseInt(r.date.slice(0, 4), 10);
    const month = parseInt(r.date.slice(5, 7), 10);
    const tzav8 = isTzav8(r.orderType);

    else y.regular = round2(y.regular + r.reserveDays);

    let ms = perMonth.get(year);
    if (!ms) perMonth.set(year, (ms = new Map()));
    let m = ms.get(month);
    if (!m) ms.set(month, (m = { key: month, tzav8: 0, regular: 0 }));
    if (tzav8) m.tzav8 = round2(m.tzav8 + r.reserveDays);
    else m.regular = round2(m.regular + r.reserveDays);

    months[String(year)] = Array.from(ms.values()).sort((a, b) => a.key - b.key);
  });

}
```

```ts
/** Date helpers shared by server and client. */

/** dd/mm/yyyy (also d.m.yy, d-m-yyyy) -> yyyy-mm-dd (empty when unparsable) */
export function ddmmyyyyToIso(s: string): string {
  const m = /^(\d{1,2})[./-](\d{1,2})[./-](\d{2,4})$/.exec((s ?? "").trim());
  if (!m) return "";
  const [, d, mo, y] = m;
  const year = y.length === 2 ? `20${y}` : y;
  const month = parseInt(mo, 10);
  const day = parseInt(d, 10);
  if (month < 1 || month > 12 || day < 1 || day > 31) return "";
  return `${year}-${String(month).padStart(2, "0")}-${String(day).padStart(2, "0")}`;
}

/** yyyy-mm-dd -> dd/mm/yyyy */
export function isoToDdmmyyyy(iso: string): string {
  const [y, m, d] = iso.split("-");
  return `${d}/${m}/${y}`;
}

/** "HH:mm" or "HH:mm:ss" -> decimal hours (0 for blank/invalid) */
export function hhmmToDecimal(s: string): number {
  const m = /^(\d{1,2}):(\d{2})(?::\d{2})?$/.exec((s ?? "").trim());
  if (!m) return 0;
  return parseInt(m[1], 10) + parseInt(m[2], 10) / 60;
}

export function decimalToHhmm(d: number): string {
  const h = Math.floor(d);
  const min = Math.round((d - h) * 60);
  return `${h}:${String(min).padStart(2, "0")}`;
}

export function daysInMonth(year: number, month: number): number {
  return new Date(year, month, 0).getDate();
}

export function todayIso(): string {
  const now = new Date();
  return `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, "0")}-${String(
    now.getDate()
  ).padStart(2, "0")}`;
}

export function round1(n: number): number {
  return Math.round(n * 10) / 10;
}
```

```ts
  VACATION_BALANCE: 27, // יתרת חופשה
  EOM_VACATION: 28, // חופשה בסוף החודש
  MAX_MONTHLY_HOURS: 29, // שעות חודשיות מקסימליות
  MONTHLY_STANDARD: 30, // תקן שעות חודשי
} as const;

export const LAST_COLUMN_LETTER = "AE";

/** Classification marking a standards-definition row, not a day report. */
export const STANDARD_ROW_CLASSIFICATION = "הגדרת תקן";

/**
 * Legacy sheets persist monthly standards as special rows at the bottom of
 * the journal: date = first of month, column E = standard work days,
 * column AE = monthly target hours. They reuse real dates (e.g. 01/07/2026),
 * so they MUST be kept out of the day-record stream.
 */
export function parseStandardRow(
  row: unknown[]
): { year: number; month: number; days: number } | null {
  if (str(row[COL.CLASSIFICATION]) !== STANDARD_ROW_CLASSIFICATION) return null;
  const iso = ddmmyyyyToIso(str(row[COL.DATE]));
  if (!iso) return null;
  const days = num(row[COL.DAILY_STANDARD_RAW]); // column E holds day count here
  if (!days) return null;
  return {
    year: parseInt(iso.slice(0, 4), 10),
    month: parseInt(iso.slice(5, 7), 10),
    days,
  };
}

function num(v: unknown): number {
  if (v === null || v === undefined || v === "") return 0;
  const n = parseFloat(String(v).replace(/[^\d.\-]/g, ""));
  return Number.isFinite(n) ? n : 0;
}

function str(v: unknown): string {
  if (v === null || v === undefined) return "";
  const s = String(v).trim();
  // Broken-formula artifacts (#REF!, #VALUE!, #N/A, #DIV/0! ...) count as empty.
  return s.startsWith("#") ? "" : s;
}

/** Parse one raw sheet row into a DayRecord. Returns null for non-date rows. */
export function parseRow(row: unknown[], rowIndex: number, tab = ""): DayRecord | null {
  const dateDisplay = str(row[COL.DATE]);
  const iso = ddmmyyyyToIso(dateDisplay);
  if (!iso) return null;

  const totalHours = str(row[COL.TOTAL_HOURS]);
  const entry = str(row[COL.ENTRY]);
  const exit = str(row[COL.EXIT]);

  // Total hours: prefer the sheet's decimal column, then the HH:mm total,
  // and finally derive from the raw entry/exit times. The last fallback keeps
  // the dashboard correct even when copied sheets lose their formulas
  // (#REF!/#VALUE! parse to 0 while the raw times survive).
  let totalDec = num(row[COL.TOTAL_HOURS_DEC]) || hhmmToDecimal(totalHours);
  if (!totalDec && entry && exit) {
    const derived = hhmmToDecimal(exit) - hhmmToDecimal(entry);
    if (derived > 0) totalDec = Math.round(derived * 100) / 100;
  }

  // Daily standard / day name: fall back to calendar-derived values when the
  // sheet cells are empty or their formulas broke (Sun-Wed 9h, Thu 8.5h).
  const dow = new Date(`${iso}T00:00:00`).getDay();
  let dailyStandard = num(row[COL.DAILY_STANDARD_DEC]) || num(row[COL.DAILY_STANDARD_RAW]);
  const classification = str(row[COL.CLASSIFICATION]);
  // Standards-definition rows reuse first-of-month dates — not day reports.
  if (classification === STANDARD_ROW_CLASSIFICATION) return null;
  const nonWorking = ['סופ"ש', "סופשבוע", "חג", "שבת"].includes(classification);
  if (!dailyStandard && !nonWorking && dow !== 5 && dow !== 6) {
    dailyStandard = dow === 4 ? 8.5 : 9;
  }
  const dayName = str(row[COL.DAY_NAME]) || HEBREW_DAYS[dow];

  return {
    rowIndex,
    tab,
    date: iso,
    dateDisplay,
    entry,
    exit,
    totalHours,
    totalHoursDecimal: totalDec,
    dailyStandard,
    overtimeDecimal:
      num(row[COL.OVERTIME_DEC]) ||
      Math.max(Math.round((totalDec - dailyStandard) * 100) / 100, 0),
    vacationDays: num(row[COL.VACATION_DAYS]),
    reserveDays: num(row[COL.RESERVE_DAYS]),
    sickDays: num(row[COL.SICK_DAYS]),
    classification,
    notes: str(row[COL.NOTES]),
    uniqueNotes: str(row[COL.UNIQUE_NOTES]),
    orderType: str(row[COL.ORDER_TYPE]),
    dayName,
    vacationBalance: num(row[COL.VACATION_BALANCE]),
    monthlyStandardHours: num(row[COL.MONTHLY_STANDARD]),
  };
}
```

```ts
/** ISO-8601 week number. */
export function isoWeek(date: Date): number {
  const d = new Date(Date.UTC(date.getFullYear(), date.getMonth(), date.getDate()));
  const dayNum = d.getUTCDay() || 7;
  d.setUTCDate(d.getUTCDate() + 4 - dayNum);
  const yearStart = new Date(Date.UTC(d.getUTCFullYear(), 0, 1));
  return Math.ceil(((d.getTime() - yearStart.getTime()) / 86400000 + 1) / 7);
}

/** Daily standard hours: Sun-Wed 9, Thu 8.5, Fri/Sat 0. */
export function dailyStandardFor(dow: number): number {
  if (dow === 5 || dow === 6) return 0;
  return dow === 4 ? 8.5 : 9;
}

/**
 * Compute the derived cells for one journal row from raw inputs, mirroring the
 * legacy sheet formulas. Overtime split observed in the source sheet:
 * first 2h at 120%, remainder at 150% (200% left for manual/holiday cases).
 */
export function computeDerivedCells(opts: {
  entry: string;
  exit: string;
  dailyStandard: number;
}): {
  totalHHMM: string;
  totalDecimal: number;
  overtime: number;
  ot120: number;
  ot150: number;
  ot200: number;
  hoursInt: number;
  minutesInt: number;
} {
  const toMin = (t: string): number => {
    const m = /^(\d{1,2}):(\d{2})$/.exec((t ?? "").trim());
    return m ? parseInt(m[1], 10) * 60 + parseInt(m[2], 10) : 0;
  };
  const totalMin = Math.max(toMin(opts.exit) - toMin(opts.entry), 0);
  const hoursInt = Math.floor(totalMin / 60);
  const minutesInt = totalMin % 60;
  const overtime = Math.max(totalMin / 60 - opts.dailyStandard, 0);
  const ot120 = Math.min(overtime, 2);
  const ot150 = Math.max(overtime - 2, 0);
  return {
    totalHHMM: `${hoursInt}:${String(minutesInt).padStart(2, "0")}`,
    totalDecimal: Math.round((totalMin / 60) * 100) / 100,
    overtime: Math.round(overtime * 100) / 100,
    ot120: Math.round(ot120 * 100) / 100,
    ot150: Math.round(ot150 * 100) / 100,
    ot200: 0,
    hoursInt,
    minutesInt,
  };
}

/**
 * Build journal rows for every date of a year (values only, no formulas).
 * Raw/report cells start empty; date-derived helper columns are pre-filled.
 */
export function buildYearRows(year: number, annualQuota: number): (string | number)[][] {
  const rows: (string | number)[][] = [];
  const dailyQuota = Math.round((annualQuota / 365) * 100) / 100;
  const start = new Date(year, 0, 1);
  for (let d = new Date(start); d.getFullYear() === year; d.setDate(d.getDate() + 1)) {
    const dow = d.getDay();
    const std = dailyStandardFor(dow);
    const weekend = std === 0;
    const dd = String(d.getDate()).padStart(2, "0");
    const mm = String(d.getMonth() + 1).padStart(2, "0");
    rows.push([
      `${dd}/${mm}/${year}`, // A תאריך
      "", // B כניסה
      "", // C יציאה
      "0:00", // D סה"כ שעות
      std, // E תקן יומי
      0, // F שעות נוספות
      0, // G 120%
      0, // H 150%
      0, // I 200%
      "", // J ימי חופש
      "", // K ימי מילואים
      "", // L ימי מחלה
      weekend ? 'סופ"ש' : "עבודה", // M סיווג
      "", // N הערות
      "", // O הערות ייחודיות
      "", // P סוג צו
      d.getMonth() + 1, // Q חודש
      year, // R שנה
      isoWeek(d), // S שבוע
      HEBREW_DAYS[dow], // T יום
      std, // U תקן יומי עשרוני
      0, // V תרגום שעות לעשרוני
      0, // W שעות נוספות עשרוני
      0, // X שעות
      0, // Y דקות
      dailyQuota, // Z מכסת חופשה יומית
      "", // AA מכסת חופשה מצטברת
      "", // AB יתרת חופשה
      "", // AC חופשה בסוף החודש
      220, // AD שעות חודשיות מקסימליות
      192, // AE תקן שעות חודשי
    ]);
  }
  return rows;
}
```

```ts
export async function readJournal(force = false): Promise<DayRecord[]> {
  const cfg = getConfig();
  const cacheKey = `${cfg.sheetId}:${cfg.sheetTab}`;
  if (!force && cache && cache.key === cacheKey && Date.now() - cache.at < CACHE_TTL_MS) {
    return cache.records;
  }
  const api = getClient();
  const tabs = await listJournalTabs();
  const res = await api.spreadsheets.values.batchGet({
    spreadsheetId: cfg.sheetId,
    ranges: tabs.map((t) => `'${t}'!A1:${LAST_COLUMN_LETTER}`),
    valueRenderOption: "FORMATTED_VALUE",
  });

  // Merge by date; a row with reported content wins over an empty duplicate.
  const byDate = new Map<string, DayRecord>();
  const standards: Record<string, number[]> = {};
  res.data.valueRanges?.forEach((vr, t) => {
    (vr.values ?? []).forEach((row, i) => {
      // Legacy standards rows: collect for settings, keep out of the journal.
      const std = parseStandardRow(row);
      if (std) {
        const y = String(std.year);
        if (!standards[y]) standards[y] = Array(12).fill(0);
        standards[y][std.month - 1] = std.days;
        return;
      }
      const rec = parseRow(row, i + 1, tabs[t]);
      if (!rec) return;
      const existing = byDate.get(rec.date);
      const hasContent =
        rec.totalHoursDecimal > 0 || rec.vacationDays > 0 || rec.sickDays > 0 || rec.reserveDays > 0;
      const existingHasContent =
        existing &&
        (existing.totalHoursDecimal > 0 ||
          existing.vacationDays > 0 ||
          existing.sickDays > 0 ||
          existing.reserveDays > 0);
      if (!existing || (hasContent && !existingHasContent)) byDate.set(rec.date, rec);
    });
  });

  const records = Array.from(byDate.values()).sort((a, b) => a.date.localeCompare(b.date));
  rowStandards = standards;
  cache = { at: Date.now(), key: cacheKey, records };
  return records;
}

 * recalculate everything else. App-created sheets (computeDerived=true): the
 * server also writes the calculated columns (D:I, V:Y).

  const r = rec.rowIndex;
  const tab = rec.tab;
  const data: sheets_v4.Schema$ValueRange[] = [
    { range: journalRange(`B${r}:C${r}`, tab), values: [[input.entry ?? "", input.exit ?? ""]] },
    {
      range: journalRange(`J${r}:N${r}`, tab),
      values: [
        [
          input.vacationDays || "",
          input.reserveDays || "",
          input.sickDays || "",
          input.classification ?? "",
          input.notes ?? rec.notes ?? "",
        ],
      ],
    },
  ];

  if (cfg.computeDerived) {
    const d = computeDerivedCells({
      entry: input.entry ?? "",
      exit: input.exit ?? "",
      dailyStandard: rec.dailyStandard,
    });
    data.push(
      {
        range: journalRange(`D${r}:I${r}`, tab),
        values: [[d.totalHHMM, rec.dailyStandard, d.overtime, d.ot120, d.ot150, d.ot200]],
      },
      {
        range: journalRange(`V${r}:Y${r}`, tab),
        values: [[d.totalDecimal, d.overtime, d.hoursInt, d.minutesInt]],
      }
    );
  }

  await api.spreadsheets.values.batchUpdate({
    spreadsheetId: cfg.sheetId,
    requestBody: { valueInputOption: "USER_ENTERED", data },
  });
  cache = null;
  const refreshed = await readJournal(true);
  return refreshed.find((x) => x.date === input.date) ?? null;
}

/** Native tabs used by legacy spreadsheets (take precedence when present). */
const NATIVE_STANDARDS_TAB = "WorkdayStandards";
const NATIVE_SETTINGS_TAB = "Settings";
const QUOTA_KEY = "AnnualVacationAllowance";

export async function readSettings(): Promise<StandardSettings> {
  const cfg = getConfig();
  // Legacy standards may live inside the journal — make sure it was scanned.
  if (!Object.keys(rowStandards).length) await readJournal();
  const api = getClient();
  const titles = await getTabTitles();

  let quota = 0;
  const years: Record<string, number[]> = {};

  const ranges: string[] = [];
  if (titles.includes(NATIVE_STANDARDS_TAB)) ranges.push(`'${NATIVE_STANDARDS_TAB}'!A1:M100`);
  if (titles.includes(NATIVE_SETTINGS_TAB)) ranges.push(`'${NATIVE_SETTINGS_TAB}'!A1:B50`);
  if (titles.includes(cfg.settingsTab)) ranges.push(`'${cfg.settingsTab}'!A1:M100`);

  const res = ranges.length
    ? await api.spreadsheets.values.batchGet({ spreadsheetId: cfg.sheetId, ranges })
    : { data: { valueRanges: [] as { values?: unknown[][] }[] } };
  const byRange = new Map<string, unknown[][]>();
  ranges.forEach((r, i) => byRange.set(r, (res.data.valueRanges?.[i]?.values ?? []) as unknown[][]));

  // 1. Native WorkdayStandards matrix (Year | Jan..Dec) — authoritative.
  const wd = byRange.get(`'${NATIVE_STANDARDS_TAB}'!A1:M100`);
  if (wd) {
    for (let i = 1; i < wd.length; i++) {
      const y = String(parseFloat(String(wd[i]?.[0] ?? "")) || "").trim();
      if (/^\d{4}$/.test(y)) {
        years[y] = Array.from({ length: 12 }, (_, m) =>
          parseFloat(String(wd[i]?.[m + 1] ?? "")) || 0
        );
      }
    }
  }

  // 2. Native Settings tab (key/value) for the vacation quota.
  const st = byRange.get(`'${NATIVE_SETTINGS_TAB}'!A1:B50`);
  if (st) {
    const row = st.find((r) => String(r?.[0] ?? "").trim() === QUOTA_KEY);
    if (row) quota = parseFloat(String(row[1] ?? "")) || 0;
  }

  // 3. App-managed tab fills whatever the native tabs didn't define.
  const app = byRange.get(`'${cfg.settingsTab}'!A1:M100`);
  if (app) {
    if (!quota) quota = parseFloat(String(app[0]?.[1] ?? "")) || 0;
    for (let i = 3; i < app.length; i++) {
      const y = String(app[i]?.[0] ?? "").trim();
      if (/^\d{4}$/.test(y) && !years[y]) {
        years[y] = Array.from({ length: 12 }, (_, m) =>
          parseFloat(String(app[i]?.[m + 1] ?? "")) || 0
        );
      }
    }
  }

  // 4. Legacy "הגדרת תקן" journal rows fill any remaining gaps.
  for (const [y, months] of Object.entries(rowStandards)) {
    const existing = years[y];
    if (!existing || existing.every((v) => !v)) years[y] = months;
  }

  return { annualVacationQuota: quota || DEFAULT_QUOTA, years };
}

export async function writeSettings(s: StandardSettings): Promise<void> {
  const cfg = getConfig();
  const api = getClient();
  const titles = await getTabTitles();
  const yearRows = Object.keys(s.years)
    .sort()
    .map((y) => [parseInt(y, 10), ...s.years[y]] as (string | number)[]);

  if (titles.includes(NATIVE_STANDARDS_TAB)) {
    // Write the matrix back in the sheet's own native format.
    await api.spreadsheets.values.update({
      spreadsheetId: cfg.sheetId,
      range: `'${NATIVE_STANDARDS_TAB}'!A1`,
      valueInputOption: "RAW",
      requestBody: {
        values: [
          ["Year", "Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"],
          ...yearRows,
        ],
      },
    });
  } else {
    await ensureSettingsTab();
    await api.spreadsheets.values.update({
      spreadsheetId: cfg.sheetId,
      range: `'${cfg.settingsTab}'!A1`,
      valueInputOption: "RAW",
      requestBody: {
        values: [
          ["מכסת חופשה שנתית", s.annualVacationQuota],
          ["", ""], // spacer must not be an empty array — the API rejects those
          ["שנה", "ינו", "פבר", "מרץ", "אפר", "מאי", "יונ", "יול", "אוג", "ספט", "אוק", "נוב", "דצמ"],
          ...yearRows,
        ],
      },
    });
  }

  // Vacation quota: prefer the native Settings key/value tab.
  if (titles.includes(NATIVE_SETTINGS_TAB)) {
    const res = await api.spreadsheets.values.get({
      spreadsheetId: cfg.sheetId,
      range: `'${NATIVE_SETTINGS_TAB}'!A1:A50`,
    });
    const idx = (res.data.values ?? []).findIndex(
      (r) => String(r?.[0] ?? "").trim() === QUOTA_KEY
    );
    if (idx >= 0) {
      await api.spreadsheets.values.update({
        spreadsheetId: cfg.sheetId,
        range: `'${NATIVE_SETTINGS_TAB}'!B${idx + 1}`,
        valueInputOption: "RAW",
        requestBody: { values: [[s.annualVacationQuota]] },
      });
    }
  } else if (titles.includes(NATIVE_STANDARDS_TAB)) {
    // Years went to the native tab; persist the quota in the app tab.
    await ensureSettingsTab();
    await api.spreadsheets.values.update({
      spreadsheetId: cfg.sheetId,
      range: `'${cfg.settingsTab}'!A1:B1`,
      valueInputOption: "RAW",
      requestBody: { values: [["מכסת חופשה שנתית", s.annualVacationQuota]] },
    });
  }
}

/** Verify an existing spreadsheet is reachable and journal-shaped. */
export async function validateExistingSheet(
  auth: GoogleAuth,
  sheetId: string
): Promise<{ title: string; firstTab: string }> {
  const api = google.sheets({ version: "v4", auth });
  const meta = await api.spreadsheets.get({ spreadsheetId: sheetId });
  const firstTab = meta.data.sheets?.[0]?.properties?.title ?? "";
  const head = await api.spreadsheets.values.get({
    spreadsheetId: sheetId,
    range: `'${firstTab}'!A1:A1`,
  });
  const a1 = String(head.data.values?.[0]?.[0] ?? "");
  if (!a1.includes("תאריך")) {
    throw new Error('הגיליון לא נראה כמו יומן דיווח (תא A1 חייב להכיל "תאריך")');
  }
  return { title: meta.data.properties?.title ?? "", firstTab };
}

/**
 * Create a brand-new journal spreadsheet for a first-time user:
 * header + a full current-year grid + settings tab, shared with the user,
 * optionally placed inside a Drive folder the user shared with the SA.
 */
export async function createNewSpreadsheet(opts: {
  auth: GoogleAuth;
  title: string;
  /** Share the sheet to this address — only needed for SA-owned sheets. */
  shareWith?: string;
  folderId?: string;
  annualQuota?: number;
}): Promise<{ sheetId: string; sheetUrl: string }> {
  const auth = opts.auth;
  const drive = google.drive({ version: "v3", auth });
  const api = google.sheets({ version: "v4", auth });
  const quota = opts.annualQuota ?? DEFAULT_QUOTA;

  // Create via Drive so an optional parent folder can be set atomically.
  const file = await drive.files.create({
    requestBody: {
      name: opts.title,
      mimeType: "application/vnd.google-apps.spreadsheet",
      ...(opts.folderId ? { parents: [opts.folderId] } : {}),
    },
    fields: "id, webViewLink",
  });
  const sheetId = file.data.id!;
  const sheetUrl =
    file.data.webViewLink ?? `https://docs.google.com/spreadsheets/d/${sheetId}/edit`;

  // Journal: header + current year.
  const year = new Date().getFullYear();
  const rows = buildYearRows(year, quota);
  await api.spreadsheets.values.update({
    spreadsheetId: sheetId,
    range: "A1",
    valueInputOption: "USER_ENTERED",
    requestBody: { values: [[...HEADER_ROW], ...rows] },
  });

  // Settings tab with default standards derived from the calendar.
  await api.spreadsheets.batchUpdate({
    spreadsheetId: sheetId,
    requestBody: {
      requests: [{ addSheet: { properties: { title: "הגדרות תקן" } } }],
    },
  });
  const monthDays = Array.from({ length: 12 }, (_, m) => {
    let c = 0;
    const days = new Date(year, m + 1, 0).getDate();
    for (let d = 1; d <= days; d++) {
      const dow = new Date(year, m, d).getDay();
      if (dow !== 5 && dow !== 6) c++;
    }
    return c;
  });
  await api.spreadsheets.values.update({
    spreadsheetId: sheetId,
    range: "'הגדרות תקן'!A1",
    valueInputOption: "RAW",
    requestBody: {
      values: [
        ["מכסת חופשה שנתית", quota],
        [],
        ["שנה", "ינו", "פבר", "מרץ", "אפר", "מאי", "יונ", "יול", "אוג", "ספט", "אוק", "נוב", "דצמ"],
        [year, ...monthDays],
      ],
    },
  });

  // SA-owned sheets must be shared so they show up in the user's account.
  // OAuth-created sheets are already owned by the user — no sharing needed.
  if (opts.shareWith) {
    await drive.permissions.create({
      fileId: sheetId,
      sendNotificationEmail: true,
      requestBody: { type: "user", role: "writer", emailAddress: opts.shareWith },
    });
  }

  return { sheetId, sheetUrl };
}
```

```js
async function walk(dir) {

    if (e.isDirectory()) out.push(...(await walk(p)));

  return out;
```

```js
#!/usr/bin/env node
/**
 * Builds a clean, secret-free distribution of the app: dist/24h-work-dashboard/
 * (+ ZIP when possible). Includes oauth-client.json so recipients get the
 * one-click Google sign-in; EXCLUDES data/ (tokens!), .env*, node_modules, .next.
 *
 * Usage: npm run package
 */
import { promises as fs } from "fs";
import path from "path";
import { execSync } from "child_process";
import { fileURLToPath } from "url";

const root = path.resolve(path.dirname(fileURLToPath(import.meta.url)), "..");
const OUT_DIR = path.join(root, "dist");
const APP_DIR = path.join(OUT_DIR, "24h-work-dashboard");

const INCLUDE_DIRS = ["app", "components", "lib", "scripts", "docs"];
const INCLUDE_FILES = [
  "package.json",
  "tsconfig.json",
  "next.config.mjs",
  "tailwind.config.ts",
  "postcss.config.mjs",
  ".env.local.example",
  ".gitignore",
  "start.bat",
  "start.sh",
  "README.md",
  "CLAUDE.md",
  "QA-REPORT.md",
  "oauth-client.json", // ships the one-click Google sign-in
];

async function copyDir(src, dst) {
  await fs.mkdir(dst, { recursive: true });
  for (const e of await fs.readdir(src, { withFileTypes: true })) {
    const s = path.join(src, e.name);
    const d = path.join(dst, e.name);
    if (e.isDirectory()) await copyDir(s, d);
    else await fs.copyFile(s, d);
  }
}

await fs.rm(OUT_DIR, { recursive: true, force: true });
await fs.mkdir(APP_DIR, { recursive: true });

let copied = 0;
for (const d of INCLUDE_DIRS) {
  try {
    await copyDir(path.join(root, d), path.join(APP_DIR, d));
    copied++;
  } catch {
    /* optional dir */
  }
}
let missingOauth = false;
for (const f of INCLUDE_FILES) {
  try {
    await fs.copyFile(path.join(root, f), path.join(APP_DIR, f));
    copied++;
  } catch {
    if (f === "oauth-client.json") missingOauth = true;
  }
}

// Safety net: never ship secrets even if include lists change later.
for (const forbidden of ["data", ".env.local", ".env", "node_modules", ".next"]) {
  await fs.rm(path.join(APP_DIR, forbidden), { recursive: true, force: true });
}

// Try to ZIP (PowerShell on Windows, zip elsewhere); folder remains either way.
let zipped = "";
const zipPath = path.join(OUT_DIR, "24h-work-dashboard.zip");
try {
  if (process.platform === "win32") {
    execSync(
      `powershell -NoProfile -Command "Compress-Archive -Path '${APP_DIR}\\*' -DestinationPath '${zipPath}' -Force"`,
      { stdio: "ignore" }
    );
  } else {
    execSync(`cd '${OUT_DIR}' && zip -qr 24h-work-dashboard.zip 24h-work-dashboard`, {
      stdio: "ignore",
      shell: "/bin/bash",
    });
  }
  zipped = zipPath;
} catch {
  /* zip tooling unavailable — folder output is enough */
}

console.log(`Distribution ready: ${APP_DIR}`);
if (zipped) console.log(`ZIP: ${zipped}`);
if (missingOauth) {
  console.warn(
    "\nWARNING: oauth-client.json not found — recipients will NOT get the one-click " +
      "Google sign-in (only the advanced Service Account mode). See README section 'הפצה'."
  );
}
```
